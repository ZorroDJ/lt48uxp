cmake_minimum_required(VERSION 3.20)
project(labtool48uxp VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(LIBUSB REQUIRED IMPORTED_TARGET libusb-1.0)
pkg_check_modules(GTK4 REQUIRED IMPORTED_TARGET gtk4>=4.10)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
 add_compile_definitions(_GNU_SOURCE)
endif()
include(core_sources.cmake)
add_library(LabTool48::core ALIAS labtool48_core)
set_target_properties(labtool48_core PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO OUTPUT_NAME labtool48 VERSION ${PROJECT_VERSION} SOVERSION 0)
target_include_directories(labtool48_core PUBLIC c/include)
target_link_libraries(labtool48_core PRIVATE PkgConfig::LIBUSB Threads::Threads)
target_compile_definitions(labtool48_core PRIVATE LABTOOL48_BUILDING_LIBRARY LT48_NEW_DEVICE_LIB_INSTALL="${CMAKE_INSTALL_FULL_DATADIR}/labtool48/NewDeviceLib")
add_executable(labtool48 c/apps/cli/main.c)
target_link_libraries(labtool48 PRIVATE LabTool48::core)
add_executable(labtool48-gtk c/apps/gtk/main.c c/apps/gtk/property_sheet.c)
target_link_libraries(labtool48-gtk PRIVATE LabTool48::core PkgConfig::GTK4 ${CMAKE_DL_LIBS})
foreach(target labtool48_core labtool48 labtool48-gtk)
 set_target_properties(${target} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO INSTALL_RPATH "$ORIGIN/../lib")
 if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 endif()
endforeach()
install(TARGETS labtool48_core labtool48 labtool48-gtk RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY c/include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES packaging/labtool48.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES packaging/70-labtool48.rules DESTINATION ${CMAKE_INSTALL_LIBDIR}/udev/rules.d)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/NewDeviceLib/library.json" new_device_index)
install(FILES NewDeviceLib/library.json NewDeviceLib/README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/labtool48/NewDeviceLib)
foreach(empty_dir bundles applets bitstreams)
 install(DIRECTORY DESTINATION "${CMAKE_INSTALL_DATADIR}/labtool48/NewDeviceLib/${empty_dir}")
endforeach()
foreach(section records resources bundles)
 string(JSON count LENGTH "${new_device_index}" ${section})
 if(count GREATER 0)
  math(EXPR last "${count}-1")
  foreach(i RANGE ${last})
   string(JSON relative GET "${new_device_index}" ${section} ${i} path)
   string(JSON expected GET "${new_device_index}" ${section} ${i} sha256)
   if(relative MATCHES "(^/|\\\\|(^|/)\\.\\.?(/|$)|:)")
    message(FATAL_ERROR "Invalid NewDeviceLib path: ${relative}")
   endif()
   file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/NewDeviceLib/${relative}" actual)
   if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "NewDeviceLib hash mismatch: ${relative}")
   endif()
   get_filename_component(parent "${relative}" DIRECTORY)
   install(FILES "NewDeviceLib/${relative}" DESTINATION "${CMAKE_INSTALL_DATADIR}/labtool48/NewDeviceLib/${parent}")
  endforeach()
 endif()
endforeach()
include(CTest)
if(BUILD_TESTING)
 add_executable(labtool48-package-test packaging/test_release.c)
 target_link_libraries(labtool48-package-test PRIVATE LabTool48::core)
 add_test(NAME package-offline COMMAND labtool48-package-test)
endif()
